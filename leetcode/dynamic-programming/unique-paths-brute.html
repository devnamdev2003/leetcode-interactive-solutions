class Solution {
    public int uniquePaths(int m, int n) {
        if (m == 1 && n == 1) {
            return 1;
        }

        int a = 0;
        int b = 0;

        if (m != 1) {
            a = uniquePaths(m - 1, n);
        }

        if (n != 1) {
            b = uniquePaths(m, n - 1);
        }

        return a + b;
    }
}
