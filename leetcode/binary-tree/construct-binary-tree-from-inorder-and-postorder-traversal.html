class Solution {
    public int k = 0;

    public TreeNode buildTree(int[] inorder, int[] postorder) {
        Map m = new HashMap<>();
        for (int i = 0; i < inorder.length; i++) {
            m.put(inorder[i], i);
        }
        return help(postorder, m, 0, inorder.length - 1);
    }

    public TreeNode help(int[] postorder, Map m, int l, int r) {
        if (l > r) {
            return null;
        }
        TreeNode root = new TreeNode(postorder[postorder.length - 1 - k++]);
        int rootIndex = m.get(root.val);
        root.right = help(postorder, m, rootIndex + 1, r);
        root.left = help(postorder, m, l, rootIndex - 1);
        return root;
    }
}
